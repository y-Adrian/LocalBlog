# 内核链表与常用数据结构

Linux 内核没有用 C++ STL，而是实现了一套精巧的 C 数据结构。理解这套 API 是读懂内核源码和写驱动的基础。

---

## 1. `container_of` —— 一切的基础

```c
/**
 * container_of(ptr, type, member)
 * 通过结构体成员的指针，反向得到外层结构体的指针
 *
 * ptr    - 指向成员的指针
 * type   - 外层结构体类型
 * member - 成员在结构体里的名字
 */
#define container_of(ptr, type, member) \
    ((type *)((char *)(ptr) - offsetof(type, member)))
```

**示例：**

```c
struct device {
    char name[32];
    struct list_head list;   /* 嵌入链表节点 */
    int id;
};

struct list_head *node = /* 从链表遍历中拿到的节点 */;

/* 通过 list 成员指针，反推 device 结构体 */
struct device *dev = container_of(node, struct device, list);

/* 等价于手动计算：*/
struct device *dev2 = (struct device *)((char *)node - offsetof(struct device, list));
```

`container_of` 是 Linux 数据结构设计哲学的核心：**把通用节点嵌入到业务数据里，而不是业务数据嵌入通用容器**。

## 2. 双向链表：`list_head`

### 2.1 原理

```c
/* list.h 里的节点定义（极简）*/
struct list_head {
    struct list_head *next, *prev;
};

/* 业务结构体里嵌入节点 */
struct my_device {
    int id;
    char name[32];
    struct list_head list;   /* 节点嵌入在业务结构里 */
};
```

链表头本身也是 `list_head`，形成循环：

```
[head] <-> [node1] <-> [node2] <-> [node3] <-> [head]
```

### 2.2 基本操作

```c
#include <linux/list.h>

/* 声明并初始化链表头（静态）*/
LIST_HEAD(my_list);

/* 或运行时初始化 */
struct list_head head;
INIT_LIST_HEAD(&head);

/* 添加节点（头插）*/
list_add(&node->list, &head);      /* 插在 head 后面（头插 → 逆序）*/
/* 添加节点（尾插）*/
list_add_tail(&node->list, &head); /* 插在 head 前面（尾插 → 顺序）*/

/* 删除节点 */
list_del(&node->list);             /* 从链表中摘除，不释放内存 */
list_del_init(&node->list);        /* 摘除后重新初始化（避免悬空指针）*/

/* 检查是否为空 */
list_empty(&head);

/* 遍历（只读）*/
struct my_device *dev;
list_for_each_entry(dev, &head, list) {
    pr_info("device: %s\n", dev->name);
}

/* 遍历（可删除节点的安全版本）*/
struct my_device *dev, *tmp;
list_for_each_entry_safe(dev, tmp, &head, list) {
    if (should_remove(dev)) {
        list_del(&dev->list);
        kfree(dev);
    }
}

/* 从第一个节点获取业务结构体 */
struct my_device *first = list_first_entry(&head, struct my_device, list);
/* 如果链表可能为空：*/
struct my_device *first = list_first_entry_or_null(&head, struct my_device, list);
```

### 2.3 驱动里的典型用法

```c
/* 管理已注册设备的链表 */
static LIST_HEAD(registered_devices);
static DEFINE_MUTEX(dev_lock);

int register_device(struct my_device *dev)
{
    mutex_lock(&dev_lock);
    list_add_tail(&dev->list, &registered_devices);
    mutex_unlock(&dev_lock);
    return 0;
}

void unregister_device(struct my_device *dev)
{
    mutex_lock(&dev_lock);
    list_del_init(&dev->list);
    mutex_unlock(&dev_lock);
}
```

## 3. 哈希链表：`hlist_head` / `hlist_node`

`list_head` 是双向循环链表，头节点和普通节点一样大。哈希表场景里，哈希槽（head）数量通常很多，希望头节点只有一个指针（节省空间），而普通节点有两个指针（支持快速删除），因此内核定义了 `hlist_head` / `hlist_node`。

```c
#include <linux/hashtable.h>

/* 定义一个 2^4 = 16 个槽的哈希表 */
DECLARE_HASHTABLE(dev_table, 4);
hash_init(dev_table);

struct my_device {
    int id;
    struct hlist_node hash;   /* 哈希链表节点 */
};

/* 插入 */
hash_add(dev_table, &dev->hash, dev->id);

/* 查找（遍历 key 对应的槽）*/
struct my_device *found = NULL;
struct my_device *dev;
hash_for_each_possible(dev_table, dev, hash, target_id) {
    if (dev->id == target_id) {
        found = dev;
        break;
    }
}

/* 删除 */
hash_del(&dev->hash);

/* 遍历所有节点 */
int bkt;
hash_for_each(dev_table, bkt, dev, hash) {
    pr_info("id=%d\n", dev->id);
}
```

## 4. 红黑树：`rb_root` / `rb_node`

内核调度器（CFS）、内存管理（VMAs）、定时器都用红黑树。驱动用红黑树较少，但遇到需要**有序查找**且数据量大时可以考虑。

```c
#include <linux/rbtree.h>

struct my_item {
    struct rb_node node;   /* 嵌入 rb_node */
    int key;
    void *data;
};

struct rb_root root = RB_ROOT;

/* 插入 */
static int rb_insert(struct rb_root *root, struct my_item *new)
{
    struct rb_node **link = &root->rb_node;
    struct rb_node *parent = NULL;

    while (*link) {
        struct my_item *cur = rb_entry(*link, struct my_item, node);
        parent = *link;
        if (new->key < cur->key)
            link = &(*link)->rb_left;
        else if (new->key > cur->key)
            link = &(*link)->rb_right;
        else
            return -EEXIST;  /* 已存在 */
    }

    rb_link_node(&new->node, parent, link);
    rb_insert_color(&new->node, root);  /* 重新着色平衡 */
    return 0;
}

/* 查找 */
static struct my_item *rb_find(struct rb_root *root, int key)
{
    struct rb_node *node = root->rb_node;
    while (node) {
        struct my_item *item = rb_entry(node, struct my_item, node);
        if (key < item->key)      node = node->rb_left;
        else if (key > item->key) node = node->rb_right;
        else                      return item;
    }
    return NULL;
}

/* 遍历（中序，有序）*/
struct rb_node *n;
for (n = rb_first(&root); n; n = rb_next(n)) {
    struct my_item *item = rb_entry(n, struct my_item, node);
    pr_info("key=%d\n", item->key);
}

/* 删除 */
rb_erase(&item->node, &root);
```

## 5. IDR/IDA —— 整数 ID 分配器

驱动需要给设备分配一个唯一整数 ID（如 `/dev/mydev0`、`/dev/mydev1`），不必自己管理 bitmap，内核提供 `idr`（ID→指针）和 `ida`（仅分配ID）。

```c
#include <linux/idr.h>

/* IDR：整数 → 指针映射 */
static DEFINE_IDR(mydev_idr);

/* 分配 ID 并存储指针 */
int id = idr_alloc(&mydev_idr, dev_ptr, 0, MAX_DEVS, GFP_KERNEL);
if (id < 0) return id;  /* -ENOSPC 等 */

/* 查找 */
struct my_dev *dev = idr_find(&mydev_idr, id);

/* 释放 */
idr_remove(&mydev_idr, id);

/* 遍历 */
int id;
void *entry;
idr_for_each_entry(&mydev_idr, entry, id) {
    struct my_dev *dev = entry;
    pr_info("id=%d\n", id);
}

/* IDA：只分配整数 ID（不存指针，更轻量）*/
#include <linux/ida.h>
static DEFINE_IDA(mydev_ida);

int id = ida_alloc(&mydev_ida, GFP_KERNEL);       /* 自动选 ID */
int id = ida_alloc_range(&mydev_ida, 0, 9, GFP_KERNEL);  /* 在 0-9 范围内 */
ida_free(&mydev_ida, id);
```

典型用法：`misc_register` 内部就用 IDA 分配 minor number。

## 6. 位图操作

内核提供了操作位图的 API，比手写位移更清晰，也经过了架构优化。

```c
#include <linux/bitmap.h>
#include <linux/bitops.h>

/* 单个 bit（在 unsigned long 上）*/
unsigned long flags = 0;
set_bit(3, &flags);    /* 置位 bit 3 */
clear_bit(3, &flags);  /* 清位 */
test_bit(3, &flags);   /* 测试 */
test_and_set_bit(3, &flags);   /* 原子 test-and-set（多 CPU 安全）*/

/* 大位图 */
DECLARE_BITMAP(my_map, 256);   /* 256 位 */
bitmap_zero(my_map, 256);
bitmap_set(my_map, 10, 5);    /* 置位 bit 10~14（共 5 位）*/
bitmap_clear(my_map, 10, 5);
int first_zero = find_first_zero_bit(my_map, 256);
int next_set   = find_next_bit(my_map, 256, 0);
```

## 7. 各数据结构对比

| 数据结构 | 插入 | 查找 | 删除 | 遍历 | 典型场景 |
|----------|------|------|------|------|----------|
| `list_head` | O(1) | O(n) | O(1) | O(n) 有序 | 设备列表、工作队列 |
| `hlist` | O(1) | O(n/槽) | O(1) | O(n) | 哈希表的桶 |
| `rb_tree` | O(log n) | O(log n) | O(log n) | O(n) 有序 | VMA、CFS、定时器 |
| `IDR` | O(log n) | O(log n) | O(log n) | O(n) | 设备号分配 |
| `bitmap` | O(1) | O(n/64) | O(1) | O(n) | IRQ 分配、cpumask |
